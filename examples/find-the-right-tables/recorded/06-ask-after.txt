SELECT SUM(plans.monthly_price) AS monthly_recurring_revenue
FROM subscriptions
JOIN plans ON subscriptions.plan_id = plans.id
WHERE (subscriptions.status IN ('A', 'P'))
  AND subscriptions.started_at <= date('now')
  AND (subscriptions.ended_at IS NULL OR subscriptions.ended_at > date('now'));
